# PyInstaller Windows version resource (VSVersionInfo).
#
# Embeds CompanyName / ProductName / FileDescription / version into the exe.
# A bare Windows executable with no version metadata is a heuristic red flag for
# antivirus engines; real software carries this resource. Wired in via the EXE()
# `version=` argument in cuflynx.spec (Windows only; ignored on Linux/macOS).
#
# The version fields must match apps/api/version.py exactly; the number that
# Windows shows in the file's Properties should not be a different release from
# the one the app reports. tests/test_version.py enforces it -- this file drifted
# to 0.1.2 while everything else said 0.1.0, which is why it is checked now.
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(0, 7, 1, 0),
    prodvers=(0, 7, 1, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,        # VOS_NT_WINDOWS32
    fileType=0x1,      # VFT_APP
    subtype=0x0,
    date=(0, 0),
  ),
  kids=[
    StringFileInfo(
      [
        StringTable(
          '040904B0',  # US English, Unicode
          [
            StringStruct('CompanyName', 'physiomelinks'),
            StringStruct('FileDescription',
                         'CUFLynx - CellML parameter exploration and analysis (sensitivity, calibration, UQ)'),
            StringStruct('FileVersion', '0.7.1.0'),
            StringStruct('InternalName', 'CUFLynx'),
            StringStruct('OriginalFilename', 'CUFLynx-windows-x86_64.exe'),
            StringStruct('ProductName', 'CUFLynx'),
            StringStruct('ProductVersion', '0.7.1.0'),
            StringStruct('LegalCopyright', 'Copyright (c) 2024-2026 Finbar J. Argus. Apache-2.0 licensed.'),
          ],
        )
      ]
    ),
    VarFileInfo([VarStruct('Translation', [1033, 1200])]),
  ],
)
